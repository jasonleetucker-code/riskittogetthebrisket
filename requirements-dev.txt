# ── Development / test dependencies ───────────────────────────────────
#
# Installed on top of the runtime manifest.  Local developers and CI
# test jobs install via ``pip install -r requirements-dev.txt`` which
# pulls in ``requirements.txt`` transitively — guaranteeing local dev
# and CI run against the same runtime + test toolchain.
#
# Anything that is ONLY needed for running the test suite (pytest,
# TestClient HTTP shim, fixtures, linters) goes here.  Production
# never installs this file.

-r requirements.txt

# Test runner.  Minor bumped 8.3 → 9.0 on 2026-04-24 to match
# what the local + production test envs actually have installed;
# the original PR pin fell behind by three days.
pytest~=9.1.1

# ``fastapi.testclient.TestClient`` is a re-export of
# ``starlette.testclient.TestClient``, which requires ``httpx`` at
# runtime since Starlette 0.21 / FastAPI 0.87.  It is NOT declared as
# a required dep by FastAPI itself — installing FastAPI alone leaves
# TestClient importable but unusable.  Pin httpx here so every test
# job has it.
httpx~=0.28.0

# Lint + format + type tooling (config in pyproject.toml).  ruff format
# is hard-gated in CI as of the A1 sweep; ruff-check and mypy gates are
# introduced report-only/ratcheted in later phases.  Production never
# installs requirements-dev.txt, so these stay out of the runtime image.
ruff~=0.6.0
mypy~=2.3.0

# tests/intel/test_workflow_scripts.py parses
# .github/workflows/intel-refresh.yml to exercise the workflow's
# embedded decision heredocs — test-only, the runtime never reads YAML.
pyyaml~=6.0
